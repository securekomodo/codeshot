// A clear name beats a clever one.
async function fetchSnippet(id) {
  const res = await fetch(`/api/snippets/${id}`);
  if (!res.ok) throw new Error("Snippet not found");

  const { language, body } = await res.json();
  return { language, body, length: body.length };
}

fetchSnippet("hello-world").then(console.log);
