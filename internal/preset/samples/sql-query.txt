WITH recent AS (
  SELECT station_id, avg(temperature) AS avg_temp
  FROM readings
  WHERE taken_at > now() - interval '24 hours'
  GROUP BY station_id
)
SELECT s.name, round(r.avg_temp::numeric, 1) AS avg_temp
FROM stations s JOIN recent r ON r.station_id = s.id
WHERE s.active
ORDER BY avg_temp DESC LIMIT 5;
